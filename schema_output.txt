--- sessions ---
CREATE TABLE sessions (
	id VARCHAR(50) NOT NULL, 
	created_at DATETIME, 
	updated_at DATETIME, 
	PRIMARY KEY (id)
)

--- feedback ---
CREATE TABLE feedback (
	id INTEGER NOT NULL, 
	session_id VARCHAR(50), 
	message_id INTEGER, 
	feedback_type VARCHAR(20) NOT NULL, 
	"query" TEXT, 
	response TEXT, 
	created_at DATETIME, 
	PRIMARY KEY (id)
)

--- cache ---
CREATE TABLE cache (
	id INTEGER NOT NULL, 
	query_hash VARCHAR(64) NOT NULL, 
	"query" TEXT NOT NULL, 
	response TEXT NOT NULL, 
	sources JSON, 
	latency FLOAT, 
	hits INTEGER, 
	created_at DATETIME, 
	PRIMARY KEY (id)
)

--- daily_stats ---
CREATE TABLE daily_stats (
	id INTEGER NOT NULL, 
	date DATE NOT NULL, 
	total_queries INTEGER, 
	cache_hits INTEGER, 
	positive_feedback INTEGER, 
	negative_feedback INTEGER, 
	PRIMARY KEY (id)
)

--- messages ---
CREATE TABLE messages (
	id INTEGER NOT NULL, 
	session_id VARCHAR(50) NOT NULL, 
	role VARCHAR(20) NOT NULL, 
	content TEXT NOT NULL, 
	sources JSON, 
	latency FLOAT, 
	cached BOOLEAN, 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(session_id) REFERENCES sessions (id)
)

--- categories ---
CREATE TABLE categories (
	id INTEGER NOT NULL, 
	name VARCHAR(50) NOT NULL, 
	display_name VARCHAR(100) NOT NULL, 
	icon VARCHAR(10), 
	created_at DATETIME, 
	PRIMARY KEY (id)
)

--- documents ---
CREATE TABLE documents (
	id INTEGER NOT NULL, 
	filename VARCHAR(255) NOT NULL, 
	original_name VARCHAR(255), 
	category_id INTEGER, 
	file_size INTEGER, 
	chunk_count INTEGER, 
	uploaded_at DATETIME, 
	indexed_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(category_id) REFERENCES categories (id)
)
